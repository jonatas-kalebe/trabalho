/* Selection sort CORRETO (indexacao 0..n-1), escrito para demonstrar o
   compilador com um algoritmo real: vetor local, passagem por referencia,
   laços aninhados, se/entao, troca de elementos. Ordena 8 inteiros. */
funcao [
	ordena(vet[]:int, n:int):int
		[ i, j, menor, tmp:int; ]
		{
			i=0;
			enquanto (i < n-1) {
				menor = i;
				j = i+1;
				enquanto (j < n) {
					se (vet[j] < vet[menor])
					entao
						menor = j;
					fimse;
					j = j+1;
				}
				tmp = vet[menor];
				vet[menor] = vet[i];
				vet[i] = tmp;
				i = i+1;
			}
			retorne 1;
		}
]
principal
	[ vet[8]:int; i:int; ]
	{
		i=0;
		enquanto (i<8) { leia vet[i]; i=i+1; }
		ordena(vet, 8);
		escreva "Ordenado: ";
		i=0;
		enquanto (i<8) { escreva vet[i]; escreva " "; i=i+1; }
		novalinha;
	}
